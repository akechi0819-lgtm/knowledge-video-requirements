cmake_minimum_required(VERSION 3.24)
project(JiangjieWindows LANGUAGES CXX)

if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Windows desktop targets require the MSVC x64 toolchain.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
add_compile_options(/utf-8 /W4 /permissive- /EHsc)

set(WEBVIEW2_SDK "" CACHE PATH "Extracted Microsoft.Web.WebView2 NuGet package")
set(JIANGJIE_MEMBER_URL "https://jiangjie.gingers.cloud/desktop-membership" CACHE STRING "Desktop membership HTTPS endpoint")
set(JIANGJIE_MEMBER_AGENT_API "" CACHE STRING "Empty for Chat Completions, or openai-responses")
set(JIANGJIE_APP_VERSION "0.2.1" CACHE STRING "Application SemVer version")
set(JIANGJIE_RELEASE_SEQUENCE 2 CACHE STRING "Strictly monotonic integer release sequence")
set(JIANGJIE_UPDATE_CHANNEL "stable" CACHE STRING "Default release update channel (stable or beta)")
set(JIANGJIE_UPDATE_FEED_URL "https://jiangjie.gingers.cloud/downloads/windows/x64/stable/feed.json" CACHE STRING "HTTPS update feed endpoint")
set(JIANGJIE_RELEASE_PUBKEY_B64 "" CACHE STRING "Optional base64url 64-byte raw ECDSA P-256 public key for signed feed verification")

if(NOT JIANGJIE_MEMBER_URL MATCHES "^https://[A-Za-z0-9.-]+(:[0-9]+)?(/[A-Za-z0-9._~/-]*)?$")
    message(FATAL_ERROR "JIANGJIE_MEMBER_URL must be a valid HTTPS service URL without credentials or query parameters.")
endif()
if(NOT JIANGJIE_MEMBER_AGENT_API MATCHES "^(openai-responses|openai-completions)?$")
    message(FATAL_ERROR "Unexpected JIANGJIE_MEMBER_AGENT_API.")
endif()
if(NOT JIANGJIE_UPDATE_CHANNEL MATCHES "^(stable|beta)$")
    message(FATAL_ERROR "JIANGJIE_UPDATE_CHANNEL must be stable or beta.")
endif()
if(NOT JIANGJIE_UPDATE_FEED_URL MATCHES "^https://[A-Za-z0-9.-]+(:[0-9]+)?(/[A-Za-z0-9._~/-]*)?$")
    message(FATAL_ERROR "JIANGJIE_UPDATE_FEED_URL must be a valid HTTPS URL.")
endif()
configure_file(build_config.h.in build_config.h @ONLY)

add_library(workbench_process STATIC workbench_process.cpp)
target_include_directories(workbench_process PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(workbench_process PUBLIC bcrypt advapi32 shell32 ole32 version)

add_library(release_engine STATIC release/release_engine.cpp)
target_include_directories(release_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/release" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(release_engine PUBLIC winhttp bcrypt wintrust advapi32 shell32 ole32)

# Core WebView2 Desktop Shell (launched by stable Launcher Jiangjie.exe)
add_executable(JiangjieApp WIN32 main.cpp)
target_include_directories(JiangjieApp SYSTEM PRIVATE "${WEBVIEW2_SDK}/build/native/include")
target_include_directories(JiangjieApp PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(JiangjieApp PRIVATE workbench_process release_engine uuid "${WEBVIEW2_SDK}/build/native/x64/WebView2LoaderStatic.lib")
target_link_options(JiangjieApp PRIVATE /MANIFESTUAC:level='asInvoker')

# Stable Entrypoint Launcher (version activation + automatic rollback + flat dev fallback)
add_executable(Jiangjie WIN32 launcher.cpp)
target_include_directories(Jiangjie PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(Jiangjie PRIVATE release_engine shell32)
target_link_options(Jiangjie PRIVATE /MANIFESTUAC:level='asInvoker')

# Background Incremental Updater Helper
add_executable(JiangjieUpdater WIN32 release/updater.cpp)
target_include_directories(JiangjieUpdater PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(JiangjieUpdater PRIVATE release_engine shell32)
target_link_options(JiangjieUpdater PRIVATE /MANIFESTUAC:level='asInvoker')

# Lightweight Online Web Bootstrapper (< 1 MiB, asInvoker, Range resume + SHA-256 + Retry UI)
add_executable(Jiangjie-Setup-Online-x64 WIN32 release/bootstrapper.cpp)
target_include_directories(Jiangjie-Setup-Online-x64 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(Jiangjie-Setup-Online-x64 PRIVATE release_engine comctl32 shell32 gdi32 user32)
target_link_options(Jiangjie-Setup-Online-x64 PRIVATE /MANIFESTUAC:level='asInvoker')

add_executable(BohackCredentialStore WIN32 credential_store.cpp)
target_link_libraries(BohackCredentialStore PRIVATE Crypt32 Shell32)
target_link_options(BohackCredentialStore PRIVATE /MANIFESTUAC:level='asInvoker')

include(CTest)
if(BUILD_TESTING)
    add_executable(workbench_process_test workbench_process_test.cpp)
    target_link_libraries(workbench_process_test PRIVATE workbench_process)
    add_test(NAME workbench_process COMMAND workbench_process_test)
    set_tests_properties(workbench_process PROPERTIES TIMEOUT 30)

    add_executable(release_engine_test release/release_engine_test.cpp)
    target_link_libraries(release_engine_test PRIVATE release_engine)
    add_test(NAME release_engine COMMAND release_engine_test)
    set_tests_properties(release_engine PROPERTIES TIMEOUT 30)
endif()

install(TARGETS Jiangjie JiangjieApp JiangjieUpdater BohackCredentialStore RUNTIME DESTINATION .)
